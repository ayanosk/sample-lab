% ---- ipsj: 情報処理学会論文誌ジャーナル（ipsj.cls）----
%
% 学会配布の ipsj.cls は pLaTeX 専用（LuaLaTeX では JT1 エンコーディングを
% 解決できず失敗する）。そのため engine と common を既定から差し替える。
% 引用も学会指定の .bst（BibTeX）に合わせて natbib 形式で出す。
%
% タイトル部は原稿の frontmatter から組む。使うキー:
%
%   title / title_en          和文・英文タイトル（改行は \\ で入れる）
%   author / author_en        著者名。「,」区切りで人数ぶん並べる
%   email                     著者のメール。「,」区切り。空欄可（例: "a@x.jp, , c@z.jp"）
%   affiliation               所属。「,」区切りで複数可。各要素は 和文\\英文
%   author_affiliation        各著者の所属番号。省略時は全員 1。
%                             現所属も持つ著者は "1,PAff1" のように書く
%   present_affiliation       現在の所属（\paffiliate）。無ければ省略してよい
%   rereceived / rerereceived 再受付・再再受付（YYYY-MM-DD。複数可）
%   abstract / abstract_en    和文・英文の要旨
%   keywords / keywords_en    和文・英文のキーワード
%   volume / number           巻・号（省略時は 0）
%   received / accepted       受付日・採録日（YYYY-MM-DD）
%
%%! fontsize     = 9pt
%%! classopts    = submit
%%! titleblock   = none
%%! bibliography = list
%%! toplevel     = default
%%! engine       = platex
%%! common       = md-common-platex.tex
%%! citations    = natbib

\documentclass[__CLASSOPTS__]{ipsj}
__COMMON__

\setcounter{巻数}{__VOLUME|0__}
\setcounter{号数}{__NUMBER|0__}
\setcounter{page}{1}

%%BEGIN-REPEAT received
\受付{__RECEIVED_Y__}{__RECEIVED_M__}{__RECEIVED_D__}
%%END-REPEAT
%%BEGIN-REPEAT rereceived
\再受付{__RERECEIVED_Y__}{__RERECEIVED_M__}{__RERECEIVED_D__}
%%END-REPEAT
%%BEGIN-REPEAT rerereceived
\再再受付{__RERERECEIVED_Y__}{__RERERECEIVED_M__}{__RERERECEIVED_D__}
%%END-REPEAT
%%BEGIN-REPEAT accepted
\採録{__ACCEPTED_Y__}{__ACCEPTED_M__}{__ACCEPTED_D__}
%%END-REPEAT

\begin{document}

\title{__TITLE__}
\etitle{__TITLE_EN__}

%%BEGIN-REPEAT affiliation
\affiliate{Aff__INDEX__}{__AFFILIATION__}
%%END-REPEAT
%%BEGIN-REPEAT present_affiliation
\paffiliate{PAff__INDEX__}{__PRESENT_AFFILIATION__}
%%END-REPEAT

%%BEGIN-REPEAT author
\author{__AUTHOR__}{__AUTHOR_EN__}{Aff__AUTHOR_AFFILIATION|1__}[__EMAIL__]
%%END-REPEAT

\begin{abstract}
__ABSTRACT__
\end{abstract}

\begin{jkeyword}
__KEYWORDS__
\end{jkeyword}

\begin{eabstract}
__ABSTRACT_EN__
\end{eabstract}

\begin{ekeyword}
__KEYWORDS_EN__
\end{ekeyword}

\maketitle

__BODY__

\bibliographystyle{ipsjunsrt}
\bibliography{__BIB__}

\end{document}
